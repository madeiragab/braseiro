# Configuracao da Mesa. Edite com o Bloco de Notas.

# Modelo usado pelo Mestre. Os numeros abaixo foram MEDIDOS nesta maquina
# (GTX 1650, 3,2 GB de VRAM utilizavel), nao estimados.
#
#  gemma3:4b   -> 7,5 tok/s | resposta longa em ~50s   | 3,3 GB  <== padrao
#                 portugues excelente, NPC com voz propria
#  qwen3:8b    -> 4,5 tok/s | ~1 min 7s               | 5,2 GB
#                 mais lento e maior, sem ganho visivel de narracao
#  mistral-nemo:12b-instruct-2407-q4_K_M
#              -> 1,55 tok/s| ~2,4 min por chamada    | 7,5 GB
#                 so 9 das 41 camadas cabem na placa. Um turno sao 3 chamadas:
#                 6 a 7 minutos por jogada. Inutilizavel aqui.
#
# A conta que manda: cada camada do 12B pesa ~175 MB e so cabem ~9 em 3,2 GB.
# Modelo menor nao e so mais rapido - ele CABE, e ai a GPU trabalha de verdade.
modelo=gemma3:4b

# Tamanho da janela de contexto. MEDIDO no Gemma 3 4B nesta placa:
#   ctx 2048 -> 8,74 tok/s   14 de 35 camadas na GPU   KV 66 MiB
#   ctx 4096 -> 8,36 tok/s   14 de 35 camadas          KV 66 MiB
#   ctx 6144 -> 7,49 tok/s   14 de 35 camadas          KV 66 MiB   <== padrao
#   ctx 8192 -> 6,48 tok/s   13 de 35 camadas          KV 60 MiB
# O Gemma 3 usa atencao de janela deslizante, entao o cache KV quase nao cresce
# com o contexto. Baixar de 6144 pra 4096 nao ganha camada nenhuma - so tira
# espaco do Mestre pra pensar. So mexa aqui se trocar de modelo.
contexto=6144

# Camadas jogadas na GPU. auto deixa o Ollama decidir.
# Na GTX 1650 (4 GB), se der erro de memoria, fixe em 16.
camadas_gpu=auto

# Criatividade. 0.8-1.0 pra RPG. Abaixo de 0.6 fica repetitivo.
temperatura=0.85

# Quantas mensagens recentes vao no contexto. O resto vive no diario e na lore.
historico=14

# Porta do painel. So mude se der conflito.
porta=11500
